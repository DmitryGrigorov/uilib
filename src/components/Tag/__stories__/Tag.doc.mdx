## Usage

```tsx
import React, { useState } from "react";
import { Tag } from "@dmitrygrigorov/components";
import { IconSetting1 } from "@dmitrygrigorov/icons";

const Example: React.FC = () => {
  const [isPressed, setIsPressed] = useState(false);
  const handleClick = (): void => {
    setIsPressed(!isPressed);
  };

  return (
    <Tag
      leadIcon={<IconSetting1 />}
      isClosable
      isPressed={isPressed}
      onClick={handleClick}>
      P2-l
    </Tag>
  );
};
```

```tsx
  const handleClick = (): void => {
    setIsPressed(!isPressed);
```

```tsx
    <Tag
      isClosable
      closeIcon={<IconCloseSquare />}

      onClick={handleClick}>
```

## Props

| Property     | Type                                                   | Default        | Required | Description                                                  |
| ------------ | ------------------------------------------------------ | -------------- | -------- | ------------------------------------------------------------ |
| children     | `string`                                               | -              | ✓        | Tag text                                                     |
| avatarProps  | `Object`                                               | -              | -        | [`Avatar`](?path=/docs/components-avatar--avatar-text) props |
| leadIcon     | `JSX.Element` // `string`                              | -              | -        | Leading icon                                                 |
| isPressed    | `boolean`                                              | `false`        | -        | Whether the tag is pressed                                   |
| isDisabled   | `boolean`                                              | `false`        | -        | Disables the tag                                             |
| closeIcon    | `JSX.Element` // `string`                              | `close-circle` | -        | Tag close icon                                               |
| isClosable   | `boolean`                                              | -              | -        | Makes the tag closable                                       |
| size         | [`TTagSize`](#ttagsize)                                | `m`            | -        | Tag size                                                     |
| className    | `string`                                               | -              | -        | Value of the `class` attribute                               |
| onClick      | `(event: React.MouseEvent<HTMLButtonElement>) => void` | -              | -        | Click handler.                                               |
| onClickCLose | `(event: React.MouseEvent<HTMLDivElement>) => void`    | -              | -        | `closeIcon` click handler.                                   |
| as           | `ElementType`                                          | -              | -        | Element used to render the tag in the DOM                    |
| isStroke     | `boolean`                                              | -              | -        | Applies the Stroke appearance                                |

:::
