## Usage

<Unstyled>
  <InputNumberExampleUsage />
</Unstyled>
<br />

```tsx
import React, { useState } from "react";
import { NumberFormatValues } from "react-number-format";
import { InputNumber } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [value, setValue] = useState<number | undefined>();

  return (
    <InputNumber
      placeholder="Example"
      value={value}
      onValueChange={(values: NumberFormatValues) => setValue(values.value)}
    />
  );
};
```

## Field content

### Placeholder

Use `placeholder` to explain what to enter or which format is expected.
`placeholder` accepts a string or a two-string array. The first label appears when the field is empty and the second when it contains a value.

<Unstyled>
  <InputNumber placeholder={["Label 1", "Label 2"]} />
</Unstyled>

```tsx
<InputNumber placeholder={["Label 1", "Label 2"]} />
```

:::

:::div

### Value

Set the field value with `value`.

<Unstyled>
  <InputNumber value={55} />
</Unstyled>

```tsx
<InputNumber value={55} />
```

:::

:::div

### Statuses

Use `status` to emphasize the input and communicate validation feedback. Available values are `error`, `warning`, and `success`.

#### status="error"

<Unstyled>
  <InputNumber
    placeholder="Error label"
    value={55}
    status="error"
    statusText="Error text"
  />
</Unstyled>

```tsx
import React, { useState } from "react";
import { InputNumber } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [value, setValue] = useState(55);

  return (
    <InputNumber
      placeholder="Error label"
      value={value}
      onChange={(_e, val: number) => setValue(val)}
      status="error"
      statusText="Error text"
    />
  );
};
```

### Step controls

Step controls increase or decrease the value by the configured increment.

<Unstyled>
  <InputNumberStepUsage />
</Unstyled>
<br />

```tsx
import React, { useState } from "react";
import { NumberFormatValues } from "react-number-format";
import InputNumber from "../../index";

const Example: React.FC = () => {
  const [value, setValue] = useState<number | undefined>();

  return (
    <InputNumber
      placeholder="Number suffix"
      value={value}
      step={1}
      onValueChange={(values: NumberFormatValues) => setValue(values.value)}
    />
  );
};
```

## Props

| Prop                | Type                                                               | Default | Required | Description                                              |
| ------------------- | ------------------------------------------------------------------ | ------- | -------- | -------------------------------------------------------- |
| value               | `string` \ `number`                                                | -       | -        | Field value                                              |
| decimalScale        | `number`                                                           | -       | -        | Number of decimal places                                 |
| type                | [`TType`](#ttype)                                                  | -       | -        | Input type                                               |
| id                  | `string`                                                           | -       | -        | Field ID in the DOM                                      |
| maxLength           | `number`                                                           | -       | -        | Maximum character count                                  |
| max                 | `number`                                                           | -       | -        | Maximum numeric value                                    |
| min                 | `number`                                                           | -       | -        | Minimum numeric value                                    |
| name                | `string`                                                           | -       | -        | Field name in the DOM                                    |
| suffix              | `string`                                                           | -       | -        | Number suffix                                            |
| thousandSeparator   | `string`                                                           | -       | -        | Thousands separator                                      |
| decimalSeparator    | `string`                                                           | -       | -        | Decimal separator                                        |
| isFixedDecimalScale | `boolean`                                                          | -       | -        | Keeps a fixed number of decimal places                   |
| step                | `number`                                                           | -       | -        | Increment used by the step controls                      |
| isAutoFocus         | `boolean`                                                          | -       | -        | Focuses and highlights the field automatically           |
| onChange            | `(event: SyntheticEvent, value: string, id?: string) => void;`     | -       | -        | Called when the field changes                            |
| onBlur              | `(event: FocusEvent<HTMLInputElement>, value: string) => void`     | -       | -        | Called when the field loses focus                        |
| onKeyPress          | `(event: KeyboardEvent<HTMLInputElement>) => void;`                | -       | -        | Handles key presses                                      |
| onValueChange       | `(values: `[`NumberFormatValues`](#numberformatvalues)`) => void;` | -       | -        | Called when the numeric value changes                    |
| viewType            | `round`, `line`                                                    | `round` | -        | Field appearance                                         |
| list                | `string`                                                           | -       | -        | Associates the field with a `<datalist>` element         |
| autoComplete        | `string`                                                           | -       | -        | Enables or disables browser autocomplete                 |
| accessKey           | `string`                                                           | -       | -        | Keyboard shortcut for activating or focusing the element |
| isReadOnly          | `boolean`                                                          | -       | -        | Makes the field read-only                                |
| form                | `string`                                                           | -       | -        | Associates the control with a form element               |
