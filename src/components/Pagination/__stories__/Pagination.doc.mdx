## Usage

Set `isShowSwitchers` to display page switchers.

```tsx
import React from "react";
import { Pagination } from "@dmitrygrigorov/components";
import { IconArrowLeft1, IconArrowRight1 } from "@dmitrygrigorov/icons";
const Example: React.FC = () => {
  const [currentPage, setCurrentPage] = React.useState(1);
  return (
    <Pagination
      currentPage={currentPage}
      onPageChange={setCurrentPage}
      totalPages={13}
      width="80vw"
      isShowSwitchers
    />
  );
};
```

## Text switchers

## Minimized pagination

A compact appearance that displays only `currentPage` and the page switchers.
Enable it with `isMinimized`.
Recommended for narrow pages and widgets.

When `totalPages >= 5` and the container is too narrow to display five page items, minimized mode is enabled automatically.

```tsx
import React from "react";
import { Pagination } from "@dmitrygrigorov/components";
import { IconArrowLeft1, IconArrowRight1 } from "@dmitrygrigorov/icons";

const Example: React.FC = () => {
  const [currentPage, setCurrentPage] = React.useState(1);
  return (
    <Pagination
      currentPage={currentPage}
      onPageChange={setCurrentPage}
      totalPages={100}
      isMinimized
    />
  );
};
```

## Custom switchers

    <Pagination
      width="130px"
      currentPage={currentPage}

      totalPages={100}

};

### Custom items-per-page values

Default values: [`defaultItemCounts`](#defaultitemcounts)
Set `paginationConf.itemsPerPage` to provide custom items-per-page choices.

```tsx
import React from "react";
import { Pagination } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [currentPage, setCurrentPage] = React.useState(1);
  return (
    <Pagination
      isShowConfig
      isShowGoToPage
      paginationConf={{
        itemsPerPage: [
          { label: "100", value: 100 },
          { label: "200", value: 200 }
        ]
      }}
      currentPage={currentPage}
      totalPages={100}
      isShowSwitchers
      width="382px"
      onPageChange={setCurrentPage}
    />
  );
};
```

### Items-per-page change handler

Use `onItemsPerPageChange` to run custom logic when the value changes.

```tsx
import React from "react";
import { Pagination } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [currentPage, setCurrentPage] = React.useState(1);

  const handleItemsPerPageChange = (value: number): void => {
    console.log(`Selected ${value} items per page`);
  };

  return (
    <Pagination
      isShowConfig
      isShowGoToPage
      paginationConf={{
        itemsPerPage: [
          { label: "100", value: 100 },
          { label: "200", value: 200 }
        ],
        onItemsPerPageChange: handleItemsPerPageChange
      }}
      currentPage={currentPage}
      totalPages={100}
      isShowSwitchers
      width="382px"
      onPageChange={setCurrentPage}
    />
  );
};
```

## Props

| Property             | Type                                        | Default | Required | Description                                                                |
| -------------------- | ------------------------------------------- | ------- | -------- | -------------------------------------------------------------------------- |
| currentPage          | `number`                                    | -       | ✓        | Current page number                                                        |
| totalPages           | `number`                                    | -       | ✓        | Total number of pages                                                      |
| onPageChange         | `(page: number) => void`                    | -       | -        | Called with the selected page number when the page changes                 |
| generalPages         | `number`                                    | 1       | -        | DEPRECATED. Will be removed in a future release. Use `totalPages` instead. |
| isShowSwitchers      | `boolean`                                   | -       | -        | Displays the switchers.                                                    |
| switcherTextNext     | `string`                                    | -       | -        | Next switcher text                                                         |
| switcherTextPrev     | `string`                                    | -       | -        | Previous switcher text                                                     |
| iconNext             | `JSX.Element`                               | -       | -        | Next switcher icon                                                         |
| iconPrev             | `JSX.Element`                               | -       | -        | Previous switcher icon                                                     |
| isDisabled           | `isDisabled`                                | -       | -        | Disables the component                                                     |
| isMinimized          | `boolean`                                   | -       | -        | Enables the minimized appearance                                           |
| isFill               | `boolean`                                   | -       | -        | Adds a filled background                                                   |
| width                | `string`                                    | -       | -        | Component width as a valid CSS value, such as `300px` or `80vw`.           |
| className            | `string`                                    | -       | -        | Additional class                                                           |
| isShowConfig         | `boolean`                                   | -       | -        | Displays the pagination configuration                                      |
| isShowGoToPage       | `boolean`                                   | -       | -        | Displays the "Go to page" field                                            |
| paginationConf       | [`IPaginationControl`](#ipaginationcontrol) | -       | -        | Pagination control configuration                                           |
| isOpenPaginationConf | `boolean`                                   | -       | -        | Controls whether the pagination configuration is open                      |
| onToggle             | `(paginationConf: boolean) => void`         | -       | -        | Called when the pagination configuration is toggled                        |

:::

:::div

### IPaginationControl

| Property             | Type                                    | Default | Required | Description                                  |
| -------------------- | --------------------------------------- | ------- | -------- | -------------------------------------------- |
| itemsPerPage         | [`IPaginationItem[]`](#ipaginationitem) | -       | -        | Available items-per-page choices.            |
| onItemsPerPageChange | `(value: number) => void`               | -       | -        | Called when the items-per-page value changes |

:::

:::div

### IPaginationItem

| Property | Type     | Default | Required | Description |
| -------- | -------- | ------- | -------- | ----------- |
| label    | `string` | -       | ✓        | Text        |
| value    | `string` | -       | ✓        | Value       |
