## Usage

The `isChecked` state indicates whether the switch is selected.

```tsx
import React, { useState } from "react";
import { Switch } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [isChecked, setIsChecked] = useState(false);
  const handleChange = (): void => {
    setIsChecked(!isChecked);
  };

  return <Switch isChecked={isChecked} onChange={handleChange}></Switch>;
};
```

## Disabled switch

Set `isDisabled` to lock the `Switch` in its current position.

```tsx
import React, { useState } from "react";
import { Switch } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [isChecked, setIsChecked] = useState(false);
  const handleChange = (): void => {
    setIsChecked(!isChecked);
  };

  return (
    <Switch
      isChecked={isChecked}
      isDisabled={isDisabled}
      onChange={handleChange}></Switch>
  );
};
```

## Text

Use `textBefore` and `textAfter` to provide custom text for each state.

```tsx
import React, { useState } from "react";
import { Switch } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [isChecked, setIsChecked] = useState(false);
  const handleChange = (): void => {
    setIsChecked(!isChecked);
  };

  return (
    <Switch
      isChecked={isChecked}
      hasTextOrIcon
      textBefore="text_2"
      textAfter="text_1"
      onChange={handleChange}></Switch>
  );
};
```

## Indeterminate switch

Set `isInderminate` to place the switch in an indeterminate state.

```tsx
export const Example: React.FC = () => {
  const [isChecked, setIsChecked] = useState(false);

  return (
    <Switch
      onChange={(_e, val) => setIsChecked(val)}
      isChecked={isChecked}
      hasTextOrIcon
      isIndeterminate
      iconAfter={<IconCloseCircle />}
      iconBefore={<IconTickCircle />}
    />
  );
};
```

::: :::div ## Props

| Property        | Type                                                   | Default | Required | Description                         |
| --------------- | ------------------------------------------------------ | ------- | -------- | ----------------------------------- |
| textAfter       | `string`                                               | -       | -        | Text for the on state               |
| textBefore      | `string`                                               | -       | -        | Text for the off state              |
| iconAfter       | `JSX.Element`                                          | -       | -        | Icon for the on state               |
| iconBefore      | `JSX.Element`                                          | -       | -        | Icon for the off state              |
| hasTextOrIcon   | `boolean`                                              | -       | -        | Displays text or icons              |
| isChecked       | `boolean`                                              | `false` | -        | Whether the switch is on            |
| isIndeterminate | `boolean`                                              | `false` | -        | Whether the switch is indeterminate |
| isDisabled      | `boolean`                                              | `false` | -        | Disables the switch                 |
| className       | `string`                                               | -       | -        | Value of the `class` attribute      |
| onChange        | `(event: React.MouseEvent<HTMLButtonElement>) => void` | -       | -        | State change handler                |
