## Usage

<Unstyled>
  <InputExampleUsage />
</Unstyled>
<br />

```tsx
import React, { useState } from "react";
import { InputDate } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [value, setValue] = useState("");

  return (
    <InputDate
      placeholder={["Label 1", "Label 2"]}
      value={value}
      onChange={(_e, val: string) => setValue(val)}
    />
  );
};
```

### Statuses

Use `status` to emphasize InputDate and communicate validation feedback. Available values are `error`, `warning`, and `success`.

#### status="error"

<Unstyled>
  <InputErrorUsage />
</Unstyled>

```tsx
import React, { useState } from "react";
import { InputDate } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [value, setValue] = useState("");

  return (
    <InputDate
      placeholder="Error label"
      value={value}
      onChange={(_e, val: string) => setValue(val)}
      status="error"
      statusText="Error text"
    />
  );
};
```

### Toggling the calendar

Set `isCalendar` to `false` to disable the calendar.

```tsx
import React, { useState } from "react";
import { InputDate } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [value, setValue] = useState("");

  return (
    <InputDate
      placeholder="Enter a date"
      isCalendar={false}
      value={value}
      onChange={(_, val: string) => setValue(val)}
    />
  );
};
```

<Unstyled>
  <OptionalInputDateOptionalCalendar />
</Unstyled>

Use `size` to change the field height and spacing.


## Props

| Prop            | Type                                                                                                                         | Default    | Required | Description                                              |
| --------------- | ---------------------------------------------------------------------------------------------------------------------------- | ---------- | -------- | -------------------------------------------------------- |
| id              | `string`, `number`                                                                                                           | -          | -        | Field ID in the DOM                                      |
| value           | `string`                                                                                                                     | -          | -        | Field value                                              |
| isAutoFill      | `boolean`                                                                                                                    | -          | -        | Enables browser autofill suggestions                     |
| minDate         | `string`                                                                                                                     | 01.01.1920 | -        | Earliest selectable date in `DD.MM.YYYY` format          |
| maxDate         | `string`                                                                                                                     | 31.12.2050 | -        | Latest selectable date in `DD.MM.YYYY` format            |
| isAutoFocus     | `boolean`                                                                                                                    | -          | -        | Focuses and highlights the field automatically           |
| name            | `string`                                                                                                                     | -          | -        | Field name in the DOM                                    |
| placeholder     | `string`, `[string, string]`                                                                                                 | -          | -        | Hint displayed inside the field                          |
| isDisabled      | `boolean`                                                                                                                    | `false`    | -        | Disables the field                                       |
| onChange        | `onChange: (event: ChangeEvent<HTMLInputElement>` <br/> `\ MouseEvent<HTMLDivElement>, value: string, id?: string) => void;` | -          | -        | Called when the field value changes                      |
| onFocus         | `event: FocusEvent<HTMLInputElement>) => void`                                                                               | -          | -        | Called when the field receives focus                     |
| onBlur          | `(event: FocusEvent<HTMLInputElement>, value: string) => void`                                                               | -          | -        | Called when the field loses focus                        |
| onKeyPress      | `(event: KeyboardEvent<HTMLInputElement>) => void;`                                                                          | -          | -        | Handles key presses                                      |
| onKeyDown       | `(event: KeyboardEvent<HTMLInputElement>) => void;`                                                                          | -          | -        | Handles key-down events                                  |
| className       | `string`                                                                                                                     | -          | -        | Additional CSS class                                     |
| testId          | `string`                                                                                                                     | -          | -        | ID used by automated tests                               |
| status          | `default`, `error`, `warning`, `success`                                                                                     | -          | -        | Field status styling                                     |
| statusText      | `string`                                                                                                                     | -          | -        | Text describing the status                               |
| isShowLabel     | `boolean`                                                                                                                    | `true`     | -        | Controls the label visibility                            |
| isShowClearIcon | `boolean`                                                                                                                    | `true`     | -        | Controls the clear button                                |
| isRangeMode     | `boolean`                                                                                                                    | `false`    | -        | Enables or disables date-range selection                 |
| viewType        | `round`, `line`                                                                                                              | `round`    | -        | Field appearance                                         |
| list            | `string`                                                                                                                     | -          | -        | Associates the field with a `<datalist>` element         |
| isCalendar      | `boolean`                                                                                                                    | `true`     | -        | Controls calendar visibility                             |
| autoComplete    | `string`                                                                                                                     | -          | -        | Enables or disables browser autocomplete                 |
| accessKey       | `string`                                                                                                                     | -          | -        | Keyboard shortcut for activating or focusing the element |
| isReadOnly      | `boolean`                                                                                                                    | -          | -        | Makes the field read-only                                |
| form            | `string`                                                                                                                     | -          | -        | Associates the control with a form element               |
