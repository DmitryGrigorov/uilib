## Usage

```tsx
import React, { useState } from "react";
import { MapObject } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [checked, setChecked] = useState(false);

  const handleChange = (event, value): void => {
    setChecked(value);
  };

  return (
    <MapObject
      isChecked={checked}
      onChange={handleChange}
      icon={<IconSetting1 />}
    />
  );
};
```

## Props

| Prop       | Type                                                  | Default | Required | Description                       |
| ---------- | ----------------------------------------------------- | ------- | -------- | --------------------------------- |
| isChecked  | `boolean`                                             | -       | -        | Whether the component is selected |
| isDisabled | `boolean`                                             | -       | -        | Whether the component is disabled |
| onClick    | `(event: React.MouseEvent<HTMLDivElement>) => void; ` | -       | -        | `onClick` event handler           |
| className  | `string`                                              | -       | -        | Additional CSS class              |
| icon       | JSX.Element                                           | -       | ✓        | Changes the icon                  |
| typeColor  | [`TMapObjectType`](#mapobjecttype)                    | -       | ✓        | Object color                      |

:::

### MapObjectType

Available values:
`danger`, `warning`, `additional`, `positive`
