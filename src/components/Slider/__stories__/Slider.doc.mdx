## Slider values

### Single value

Pass a number to `value` to let the user select a single value.

```tsx
import { useState } from "react";
import { Slider } from "@dmitrygrigorov/components";

const SliderExample = () => {
  const [value, setValue] = useState(25);

  return (
    <Slider
      leadText={`Value ${value}`}
      value={value}
      onChange={(_, val) => setValue(val)}
    />
  );
};
```

### Range

To select a range, pass a two-number array to `value` and enable `isRange`. With TypeScript, also pass `true` as the component generic to indicate range mode.

```tsx
import { useState } from "react";
import { Slider } from "@dmitrygrigorov/components";

export const SliderExampleBasicRange = () => {
  const [value, setValue] = useState<[number, number]>([20, 70]);

  return (
    <Slider<true>
      leadText={`Value ${value[0]} - ${value[1]}`}
      value={value}
      onChange={(_, val) => setValue(val)}
      isRange
    />
  );
};
```

### Minimum and maximum

      min={20}
      max={55}

  );
};

### Value constraints

Use `step` to constrain the selectable values to an interval or an explicit set.

### Step interval

Pass an integer to `step` to move the thumb between values separated by that interval.

```tsx

export const SliderExampleStepSingle: React.FC = () => {
  const [value, setValue] = useState(20);

  return (
    <Slider
      leadText={`Value ${value}`}
      value={value}
      onChange={(_, val) => setValue(val)}
      step={10}
    />
  );
};
```

## Labels and tooltips

### Tooltip

The tooltip displays each thumb's value and is enabled by default. Set `isTooltip={false}` to hide it.

```tsx
import React, { useState } from "react";
import { Slider } from "@dmitrygrigorov/components";

export const SliderExampleWithoutTooltip: React.FC = () => {
  const [value, setValue] = useState(20);

  return (
    <Slider
      leadText={`Value ${value}`}
      value={value}
      onChange={(_, val) => setValue(val)}
      isTooltip={false}
    />
  );
};
```

<Unstyled>
  <SliderExampleWithoutTooltip />
</Unstyled>
<br />

### Custom tooltip

Format the tooltip text with `tooltipFormatter`.

```tsx
import React, { useState } from "react";
import { Slider } from "@dmitrygrigorov/components";

export const SliderExampleTooltipFormatter: React.FC = () => {
  const [value, setValue] = useState(20);

  return (
    <Slider
      leadText={`Value ${value}`}
      value={value}
      onChange={(_, val) => setValue(val)}
      tooltipFormatter={(val) => `${val || 0}%`}
    />
  );
};
```

## Props

| Property         | Type                                            | Default      | Required | Description                        |
| ---------------- | ----------------------------------------------- | ------------ | -------- | ---------------------------------- |
| direction        | `horizontal`, `vertical`                        | `horizontal` | -        | Slider orientation                 |
| isDisabled       | `boolean`                                       | `false`      | -        | Disables the component             |
| value            | `TSliderValue<Range>`                           | -            | ✔       | Current value                      |
| isDisabled       | `boolean`                                       | `false`      | -        | Disables the field                 |
| onChange         | `TOnChangeSlider<Range>`                        | -            | ✔       | Handles value changes              |
| className        | `string`                                        | -            | -        | Additional CSS class               |
| status           | `info`, `error`, `warning`, `success`           | -            | -        | Visual status                      |
| width            | `string`                                        | 100%         | -        | Component width                    |
| height           | `string`                                        | -            | -        | Component height                   |
| size             | `m`, `s`                                        | `m`          | -        | Component size                     |
| step             | `number` <code>&#124;</code> `[number, number]` | 1            | -        | Step interval or selectable values |
| isRange          | `boolean`                                       | `false`      | -        | Enables range selection            |
| isTooltip        | `boolean`                                       | `true`       | -        | Shows value tooltips               |
| tooltipFormatter | `TSliderTooltipFormatter`                       | -            | -        | Formats tooltip text               |
| min              | `number`                                        | 0            | -        | Minimum value                      |
| max              | `number`                                        | 100          | -        | Maximum value                      |
| leadText         | `string`                                        | -            | -        | Leading text                       |
| trailText        | `string`                                        | -            | -        | Trailing text                      |
| defaultValue     | `TSliderValue<Range>`                           | 0            | -        | Reset value                        |
| isReset          | `boolean`                                       | `false`      | -        | Shows the reset control            |

:::

:::div

```ts
export type TSliderValue<Range> = Range extends true
  ? [number, number]
  : number;
```

```ts
export type TOnChangeSlider<Range> = (
  event: Event | TouchEvent | MouseEvent | KeyboardEvent | ChangeEvent | null,
  value: TSliderValue<Range>
) => void;
```
