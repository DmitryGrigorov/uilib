## List content

Pass dropdown items through the `options` array. By default, each option object contains these fields:

- `value: string` - unique option value
- `label: string` - option label

```tsx
const options = [
  {
    label: "One",
    value: "one"
  },
  {
    label: "Two",
    value: "two"
  }
];
```

You can also provide custom option shapes by configuring [`fieldNames`](#using-custom-options).

### Using custom options

For custom `options`, pass a `fieldNames` object that points to the value and label keys in each option.

```tsx
import React, { useState } from "react";
import { Select } from "@dmitrygrigorov/components";

const options = [
  {
    id: "Moscow",
    country: {
      city: "Moscow"
    }
  },
  {
    id: "Paris",
    country: {
      city: "Paris"
    }
  },
  {
    id: "Prague",
    country: {
      city: "Prague"
    }
  },
  {
    id: "Amsterdam",
    country: {
      city: "Amsterdam"
    },
    isDisabled: true
  }
];

const SelectFieldNamesExample: React.FC = () => {
  const [value, setValue] = useState("");

  return (
    <Select
      placeholder="Select a city"
      value={value}
      onChange={(_e, val: string) => setValue(val)}
      options={options}
      fieldNames={{ value: "id", label: "country.city" }}
    />
  );
};
```

## Placeholders

Use `placeholder` for the text shown when no option is selected. It accepts either a string or a two-item string array. With an array, the second value is shown when the field is focused or has a selected option.

```tsx
const options = [
  { label: "Moscow", value: "Moscow" },
  { label: "Paris", value: "Paris" },
  { label: "Prague", value: "Prague" },
  { label: "Amsterdam", value: "Amsterdam", isDisabled: true },
  { label: "Berlin", value: "Berlin" },
  { label: "London", value: "London" },
  { label: "Riga", value: "Riga" }
];

import React, { useState } from "react";
import { Select } from "@dmitrygrigorov/components";

const SelectPlaceholderExample: React.FC = () => {
  const [value, setValue] = useState("");

  return (
    <Select
      placeholder={["Select a city", "City"]}
      value={value}
      onChange={(_e, val: string) => setValue(val)}
      options={options}
    />
  );
};
```

## TypeScript support

Provide the option type to `Select`. For custom option shapes, also provide the string paths to the `value` and `label` fields as generic parameters.

```tsx
interface IOption {
  label: string;
  city: string;
}

const options = [
  { label: "Moscow", city: "Moscow" },
  { label: "Paris", city: "Paris" },
  { label: "Prague", city: "Prague" },
  { label: "Amsterdam", city: "Amsterdam", isDisabled: true },
  { label: "Berlin", city: "Berlin" },
  { label: "London", city: "London" },
  { label: "Riga", city: "Riga" }
];

import React, { useState } from "react";
import { Select } from "@dmitrygrigorov/components";

const SelectPlaceholderExample: React.FC = () => {
  const [value, setValue] = useState("");

  return (
    <Select<IOption, "city", "label">
      placeholder={["Select a city", "City"]}
      value={value}
      onChange={(_e, val: string) => setValue(val)}
      options={options}
      fieldNames={{ label: "label", value: "city" }}
    />
  );
};
```

## Avatars

For user selection, set `type="avatar"` to display avatars in both the menu and the field.

```tsx
const optionsUsers = [
  { label: "User1", value: "User1" },
  {
    label: "User2",
    value: "User2",
    url: AvatarImg
  },
  { label: "User3", value: "User3", url: AvatarImg },
  { label: "User4", value: "User4" }
];

import React, { useState } from "react";
import { Select } from "@dmitrygrigorov/components";

const SelectAvatarExample: React.FC = () => {
  const [value, setValue] = useState("");

  return (
    <Select
      placeholder="Select users"
      value={value}
      onChange={(_e, val: string) => setValue(val)}
      options={optionsUsers}
      type="avatar"
    />
  );
};
```

| onFocus         | `(event: FocusEvent<HTMLSelectElement>) => void`                                                                              | -                                    | -        | Handles focus                                                                                                         |

| iconLeft        | `React.ReactNode`                                                                                                             | -                                    | -        | Left icon                                                                                                             |
| type            | `basic`, `avatar`                                                                                                             | basic                                | -        | Enables basic or avatar options                                                                                       |
