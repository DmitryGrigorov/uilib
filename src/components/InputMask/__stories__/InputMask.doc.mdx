## Usage

Example using a date mask.

```tsx
import React, { useState } from "react";
import { InputMask } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [value, setValue] = useState("");

  return (
    <InputMask
      placeholder={"Date input example"}
      value={value}
      mask={Date}
      onChange={(_e, val: string) => setValue(val)}
      isShowMask={true}
      placeholderChar="_"
    />
  );
};
```

## Field type

The `type` prop selects the field type.

- `text`
- `password`
- `tel`
- `number`
- `date`
- `file`
- `email`
- `url`

## Field content

### Placeholder

Use `placeholder` to explain what to enter or which format is expected.
`placeholder` accepts a string or a two-string array. The first label appears when the field is empty and the second when it contains a value.

```tsx
export const InputMaskExample = () => {
  return <InputMask mask={Date} placeholder={["Label 1", "Label 2"]} />;
};
```

## Masks

### Regular expressions

The following example uses a regular expression to accept digits only.

```tsx
export const InputMaskRegexpExample: React.FC = () => (
  <InputMask
    width="400px"
    mask={/^\d+$/}
    size="l"
    placeholder="Regular expression"
  />
);
```

<Unstyled>
  <InputMaskRegexpExample />
</Unstyled>
<br />

#### Note

`Ensure the regular expression accepts every intermediate result, not only the final value.`

For example:

```tsx
export const InputMaskRegexpExample: React.FC = () => (
  <InputMask
    width="400px"
    mask={/^123$/}
    size="l"
    placeholder="Regular expression"
  />
);
```

This example prevents all input because only the complete string `123` matches, while intermediate values `1` and `12` do not.

### Function

This function mask accepts any ascending sequence from 0 to 9.

```tsx
export const InputMaskFunctionExample: React.FC = () => {
  const mask = (value: string): boolean =>
    /^\d*$/.test(value) &&
    value.split("").every((ch, i) => {
      const prevCh = value[i - 1];
      return !prevCh || prevCh < ch;
    });

  return (
    <InputMask
      width="400px"
      mask={mask}
      size="l"
      placeholder="Sequence from 0 to 9"
    />
  );
};
```

### String

The string can contain the following mask definitions:

- 0 — any digit
- a — any letter
- - — any character
- [] — optional input
- {} — include the fixed part in the unmasked value

`Escape a definition character when it should be treated as a fixed character.`

```tsx
export const InputMaskStringCardExample: React.FC = () => (
  <InputMask
    width="400px"
    mask="0000 0000 0000 0000"
    size="l"
    placeholder="Bank card"
    isShowMask
  />
);
```

### Complex masks

Use the `blocks` option for complex masks.

```tsx
import { InputMaskTypes } from "@dmitrygrigorov/components";

export const InputMaskContractNumberExample: React.FC = () => (
  <InputMask
    width="400px"
    mask="Contract no. NN dated DD.MM.2\02Y"
    size="l"
    placeholder="Contract"
    isShowMask
    blocks={{
      NN: {
        mask: "**"
      },
      DD: {
        mask: InputMaskTypes.MaskedRange,
        from: 1,
        to: 31
      },
      MM: {
        mask: InputMaskTypes.MaskedRange,
        from: 1,
        to: 12
      },
      Y: {
        mask: "0"
      }
    }}
  />
);
```

| className         | `string`                                                                                                                     | -       | -        | Additional CSS class                                                                                                |
| testId            | `string`                                                                                                                     | -       | -        | ID used by automated tests                                                                                          |

| viewType          | `round`, `line`                                                                                                              | `round` | -        | Field appearance                                                                                                    |
| list              | `string`                                                                                                                     | -       | -        | Associates the field with a `<datalist>` element                                                                    |
| autoComplete      | `string`                                                                                                                     | -       | -        | Enables or disables browser autocomplete                                                                            |
| accessKey         | `string`                                                                                                                     | -       | -        | Keyboard shortcut for activating or focusing the element                                                            |
| isReadOnly        | `boolean`                                                                                                                    | -       | -        | Makes the field read-only                                                                                           |
| form              | `string`                                                                                                                     | -       | -        | Associates the control with a form element                                                                          |
| blocks            | `object`                                                                                                                     | -       | -        | Configuration for complex masks                                                                                     |
