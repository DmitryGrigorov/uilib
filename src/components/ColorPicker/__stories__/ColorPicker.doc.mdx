## Usage

Pass `color` to initialize the HEX value and `onChange` to update it.

### TypeScript
```tsx
import React, { useState } from "react";
import { ColorPicker } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [color, setColor] = useState("#8bd046");

  <ColorPicker
    color={color}
    onChange={({ color: _color }) => setColor(_color)}
  />
};
```

## Adding user colors

Pass an array through `lastColors` to show colors the user selected recently.

```tsx
import React, { useState } from "react";
import { ColorPicker } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [color, setColor] = useState("#8bd046");

  <ColorPicker
    color={color}
    onChange={({ color: _color }) => setColor(_color)}
    lastColors={[
      "#6fc74c",
      "#5BB9FF",
      "#835b11",
      "#ff5bd1",
      "#3b0a08",
      "#C7514C",
      "#628141",
      "#4c56c7"
    ]}
  />
};
```

## Adding an input

Use `isInput` to add a text field that accepts a HEX color string.

```tsx
import React, { useState } from "react";
import { ColorPicker } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [color, setColor] = useState("#8bd046");

  return (
    <ColorPicker
      color={color}
      isInput={true}
      onChange={({ color: _color }) => setColor(_color)}
    />
  );
};
```

## Adding an alpha channel

Use `isTransparency` to expose the alpha channel. Set its initial value with `alpha`, which accepts a number from 0 to 1.

```tsx
import React, { useState } from "react";
import { ColorPicker } from "@dmitrygrigorov/components";

const Example: React.FC = () => {
  const [color, setColor] = useState("#8bd046");
  const [alpha, setAlpha] = useState(0.1);

  return (
    <ColorPicker
     color={color}
     isInput
     onChange={({ color: _color, alpha: _alpha }) => {
       setColor(_color);
       _alpha && setAlpha(_alpha);
     }}
     isTransparency
     alpha={alpha}
    />
  );
};
```

## Properties

| Parameter          | Type                                                 | Default              | Required | Description                                                                                       |
|--------------------|------------------------------------------------------|----------------------|----------------|--------------------------------------------------------------------------------------------------------------------------|
| className          | `string`                                             | -                    | -        | Additional class                                                                                  |
| id                 | `string`                                             | -                    | -        | Field ID in the DOM                                                                                |
| color              | `string`                                             | -                    | ✓        | Initial palette color in HEX format                                                               |
| lastColors         | `string[]`                                           | -                    | -        | Previously used HEX colors                                                                         |
| onChange           | `({ color: string, alpha?: number }) => void`        | -                    | -        | Updates the `color` value                                                                           |
| isInput            | `boolean`                                            | -                    | -        | Shows the color input                                                                               |
| colorKey           | `(index: number, color: string) => string` \| number | -                    | -        | Generates IDs for `lastColors`; without it, the color value is used as the ID                       |
| alpha              | `boolean`                                            | 1                    | -        | Initial alpha-channel value from `0` to `1`                                                         |
| isTransparency     | `boolean`                                            | -                    | -        | Shows the alpha channel                                                                             |
| testId             | `string`                                             | `color-picker`       | -        | Value of the `data-testid` attribute                                                                |
| testIdInputColor   | `string`                                             | `color-picker-input` | -        | `testId` value for the color input                                                                  |
| testIdTransparency | `string`                                             | `transparency-input` | -        | `testId` value for the alpha-channel input                                                          |
