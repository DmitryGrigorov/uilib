# Header

The table's leading section with additional controls.

## Text

Use `title` to set the table title.
Use `description` to add supporting text.

```tsx
export const HeaderTable: React.FC<IHeaderTable> = (props) => {
  const columns = useMemo(
    () => [
      {
        field: "postId"
      },
      {
        field: "id"
      },
      {
        field: "name"
      },
      {
        field: "email"
      }
    ],
    []
  );

  const [comments, setComments] = useState<any[]>([]);

  const loadData = async (): Promise<void> => {
    const response = await fetch(
      "https://jsonplaceholder.typicode.com/comments?_start=0&_limit=10"
    );
    const data = (await response.json()) as any[];
    setComments(data);
  };

  useEffect(() => {
    loadData();
  }, []);

  return (
    <Table
      title="Title"
      description="Description"
      size="m"
      columns={columns}
      rowData={comments}
    />
  );
};
```

</Unstyled>

### Refreshing the table

Pass `isRefreshable` and an `onUpdate` function to define table refresh behavior.
When `isRefreshable` is `true`, a refresh button appears in the header.

```tsx

  const columns = useMemo(
    () => [
      {
        field: "postId"
      },
      {
        field: "id"
      },
      {
        field: "name"
      },
      {
        field: "email"
      }
    ],
    []
  );

  const [comments, setComments] = useState<any[]>([]);

  const loadData = async (): Promise<void> => {
    const response = await fetch(
      "https://jsonplaceholder.typicode.com/comments?_start=0&_limit=10"
    );
    const data = (await response.json()) as any[];
    setComments(data);
  };

  useEffect(() => {
    loadData();
  }, []);

  return (
    <Table
      title="Title"
      size="m"
      columns={columns}
      rowData={comments}
      isRefreshable
      onUpdate={() => alert("Imagine this refreshes the table")}
    />
  );
};
```

## Properties

| Property                | Type          | Description                                          |
| ----------------------- | ------------- | ---------------------------------------------------- |
| title                   | string        | Header text                                          |
| description             | string        | Description text                                     |
| isRefreshable           | boolean       | Enables table refresh                                |
| headerContent           | `JSX.Element` | Header content                                       |
| additionalButton        | `JSX.Element` | Additional button content                            |
| onClickAdditionalButton | `() => void`  | Callback fired when the additional button is clicked |
| onUpdate                | `() => void`  | Callback fired when the refresh button is clicked    |
