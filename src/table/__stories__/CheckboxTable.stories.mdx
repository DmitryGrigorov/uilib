# Row selection

Table rows can be selected with checkboxes.
Pass `isCheckboxSelection` to the table to enable this behavior.
A checkbox then appears in every row.

```tsx
export const CheckboxTable: React.FC = (props) => {
  const columns: TColumn[] = useMemo(
    () => [
      {
        field: "postId"
      },
      {
        field: "id"
      },
      {
        field: "name"
      },
      {
        field: "email"
      }
    ],
    []
  );

  const [comments, setComments] = useState<any[]>([]);

  const loadData = async (): Promise<void> => {
    const response = await fetch(
      "https://jsonplaceholder.typicode.com/comments?_start=0&_limit=10"
    );
    const data = (await response.json()) as any[];
    setComments(data);
  };

  useEffect(() => {
    loadData();
  }, []);

  return (
    <Table
      size="m"
      title="Title"
      columns={columns}
      isCheckboxSelection
      rowData={comments}
      {...props}
    />
  );
};
```

### Track changes

Use the `onRowSelected` callback to track row-selection changes.

```tsx
export const CheckboxTable: React.FC = (props) => {
  const columns: TColumn[] = useMemo(
    () => [
      {
        field: "postId"
      },
      {
        field: "id"
      },
      {
        field: "name"
      },
      {
        field: "email"
      }
    ],
    []
  );

  const [comments, setComments] = useState<any[]>([]);

  const loadData = async (): Promise<void> => {
    const response = await fetch(
      "https://jsonplaceholder.typicode.com/comments?_start=0&_limit=10"
    );
    const data = (await response.json()) as any[];
    setComments(data);
  };

  useEffect(() => {
    loadData();
  }, []);

  return (
    <Table
      size="m"
      title="Title"
      columns={columns}
      isCheckboxSelection
      onRowSelected={({ node }) => alert("Row changed: " + node.id)}
      rowData={comments}
      {...props}
    />
  );
};
```

### Control which rows can be selected

Use the `isRowSelectable` callback to define whether a row can be selected.

```tsx
export const CheckboxTable: React.FC = (props) => {
  const columns: TColumn[] = useMemo(
    () => [
      {
        field: "postId"
      },
      {
        field: "id"
      },
      {
        field: "name"
      },
      {
        field: "email"
      }
    ],
    []
  );

  const [comments, setComments] = useState<any[]>([]);

  const loadData = async (): Promise<void> => {
    const response = await fetch(
      "https://jsonplaceholder.typicode.com/comments?_start=0&_limit=10"
    );
    const data = (await response.json()) as any[];
    setComments(data);
  };

  useEffect(() => {
    loadData();
  }, []);

  return (
    <Table
      size="m"
      title="Title"
      columns={columns}
      isCheckboxSelection
      isRowSelectable={(node) => {
        if (node?.id === "row-2") {
          return false;
        }
        return true;
      }}
      rowData={comments}
      {...props}
    />
  );
};
```

## Properties

| Property                  | Type                                              | Description                                                                                                 |
| ------------------------- | ------------------------------------------------- | ----------------------------------------------------------------------------------------------------------- |
| isCheckboxSelection       | `boolean`                                         | Enables row selection                                                                                       |
| isHeaderCheckboxSelection | `boolean`                                         | Displays one header checkbox for all rows                                                                   |
| onRowSelected             | `(event: IRowSelectedEvent<TData>) => void`       | Callback fired when a row's selection changes                                                               |
| isRowSelectable           | `(node: IRowNode<TData>) => boolean`              | Determines which rows can be selected. Rows are selectable by default; return `false` to prevent selection. |
| onSelectionChanged        | `(event: ISelectionChangedEvent<TData>) => void;` | Callback fired when the selection changes; returns the selected row parameters                              |

### IRowSelectedEvent

```tsx
export interface IRowSelectedEvent<TData> {
  data?: TData;
  node: IRowNode<TData>;
  rowIndex: number | null;
  rowPinned: TRowPinnedType;
  event?: Event | null;
  eventPath?: EventTarget[];
  isSelected: boolean;
}
```

### ISelectionChangedEvent

```tsx
export interface ISelectionChangedEvent<TData> {
  rowsNodes: IRowNode<TData>[];
}
```

## API methods

| Property            | Type                                                       | Description                                         |
| ------------------- | ---------------------------------------------------------- | --------------------------------------------------- |
| getSelectedNodes    | `() => IRowNode<TData>[]`                                  | Returns all selected row nodes                      |
| getSelectedRows     | `() => TData[]`                                            | Returns all selected rows                           |
| selectAll           | `() => void`                                               | Selects all rows                                    |
| deselectAll         | `() => void`                                               | Deselects all rows                                  |
| selectAllFiltered   | `() => void`                                               | Selects all filtered rows                           |
| deselectAllFiltered | `() => void`                                               | Deselects all filtered rows                         |
| setNodesSelected    | `(params: {nodes: IRowNode[]; newValue: boolean}) => void` | Applies the specified value to every provided node. |
