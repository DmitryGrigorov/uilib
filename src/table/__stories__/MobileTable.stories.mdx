# Tables for mobile devices

A table designed for mobile and responsive applications.

## Enable the mobile view

Set `isMobile: true` to enable the mobile table layout.

```tsx
export const MobileTableExample: React.FC<IHeaderTable> = (props) => {
  const columns = useMemo(
    () => [
      {
        field: "postId"
      },
      {
        field: "id"
      },
      {
        field: "name"
      },
      {
        field: "email"
      }
    ],
    []
  );

  const [comments, setComments] = useState<any[]>([]);

  const loadData = async (): Promise<void> => {
    const response = await fetch(
      "https://jsonplaceholder.typicode.com/comments?_start=0&_limit=10"
    );
    const data = (await response.json()) as any[];
    setComments(data);
  };

  useEffect(() => {
    loadData();
  }, []);

  return (
    <Table
      isSearch
      isMobile
      isPagination
      isPaginationRowPerPage
      size="s"
      title="Title"
      columns={columns}
      rowData={comments}
      {...props}
    />
  );
};
```
