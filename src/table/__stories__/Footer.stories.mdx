# Footer

The table footer provides additional controls and can combine several elements in any order.

## Pagination

Set `isPagination` to display page controls. Add `isPaginationRowPerPage` when users should be able to choose how many items appear per page. See the [pagination documentation](?path=/docs/tables-pagination--documentation) for details.

## Search

The footer can include a table search field. Set `isSearch` and `positionSearch="footer"`. See the [search documentation](?path=/docs/tables-global-table-search--documentation) for details.

## Text

Use `footerText` to add custom text to the footer.

```tsx
export const FooterTable: React.FC<IFooterTable> = (props) => {
  const columns = useMemo(
    () => [
      {
        field: "postId"
      },
      {
        field: "id"
      },
      {
        field: "name"
      },
      {
        field: "email"
      }
    ],
    []
  );

  const [comments, setComments] = useState<any[]>([]);

  const loadData = async (): Promise<void> => {
    const response = await fetch(
      "https://jsonplaceholder.typicode.com/comments?_start=0&_limit=10"
    );
    const data = (await response.json()) as any[];
    setComments(data);
  };

  useEffect(() => {
    loadData();
  }, []);

  return (
    <Table
      title="Title"
      size="m"
      columns={columns}
      rowData={comments}
      footerText={"Footer text"}
    />
  );
};
```

## Additional content

The footer can also contain custom content, such as a button with application-specific behavior.

```tsx
export const FooterTable: React.FC<IFooterTable> = (props) => {
  const columns = useMemo(
    () => [
      {
        field: "postId"
      },
      {
        field: "id"
      },
      {
        field: "name"
      },
      {
        field: "email"
      }
    ],
    []
  );

  const [comments, setComments] = useState<any[]>([]);

  const loadData = async (): Promise<void> => {
    const response = await fetch(
      "https://jsonplaceholder.typicode.com/comments?_start=0&_limit=10"
    );
    const data = (await response.json()) as any[];
    setComments(data);
  };

  useEffect(() => {
    loadData();
  }, []);

  return (
    <Table
      title="Title"
      size="m"
      columns={columns}
      rowData={comments}
      footerContent={<Button size="m">Button</Button>}
    />
  );
};
```

## Properties

| Property      | Type               | Description        |
| ------------- | ------------------ | ------------------ |
| text          | string             | Footer text        |
| footerContent | React.ReactElement | Additional content |
