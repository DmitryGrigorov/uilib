# Cell types

Tables support several cell types.

## Usage

Set `columnTypes: <cellType>` in a column definition to choose a cell type. When omitted, the table uses `basic`.

## `basic` cell type

The `basic` cell is the general-purpose default. Use `cellIcon` to display an icon on the left and `cellTrailContent` to display trailing content on the right.

```tsx
import React from "react";
import Table from "@dmitrygrigorov/table";
import { TColumn } from "@dmitrygrigorov/table";

export const TableCells: React.FC = () => {
  const columns: TColumn[] = useMemo(
    () => [
      {
        field: "id",
        title: "id"
      },
      {
        field: "fullName",
        title: "Full name",
        cellIcon: <IconSetting1 />
      },
      {
        field: "email",
        title: "email",
        cellTrailContent: <IconSetting1 />
      }
    ],
    []
  );

  const row = [
    {
      id: "1",
      fullName: "Palmerin de Lonfal",
      email: "palmslonfale@mail.com"
    },
    {
      id: "2",
      fullName: "Damien de la Tour",
      email: "dmdtr@mail.com"
    },
    {
      id: "3",
      fullName: "Auguste de Montferrand",
      email: "ogmonfer@fmail.com"
    }
  ];

  return <Table title="Cells" columns={columns} rowData={row} />;
};
```

## `status` cell type

Set `columnTypes: "status"` in the column definition to use status cells. Configure their status-specific behavior with `cellParamsGetter`.

#### `cellParamsGetter`

`cellParamsGetter` can be a function that receives `TCellStatusParamsGetterParams`, allowing each status cell to be configured independently. When all cells share the same properties, pass a `TCellParamsStatus` object instead.

```tsx
type TCellStatusParamsGetterParams<TData = any, TValue = string> = {
  value: TValue | null | undefined;
  node: IRowNode<TData> | null;
  data: TData | undefined;
  column: TColumn<TData, TValue>;
};

interface IColumnStatus<TData = any, TValue = string>
  extends Omit<IColumnBasic<TData, TValue>, "columnTypes"> {
  columnTypes: "status";
  cellParamsGetter: TCellParamsStatusGetter | TCellParamsStatus;
}

type TCellParamsStatusGetter<TData = any, TValue = string> = (
  params: TCellStatusParamsGetterParams<TData, TValue>
) => TCellParamsStatus;

export type TCompStatus =
  | "processing"
  | "completed"
  | "warning"
  | "success"
  | "error";

type TCellParamsStatus = {
  status: TCompStatus;
  leadIcon?: JSX.Element;
  trailIcon?: JSX.Element;
  isFilled?: boolean;
};
```

```tsx
import React from "react";
import Table from "@dmitrygrigorov/table";
import { TCellParamsStatus, TColumn } from "@dmitrygrigorov/table";

export const TableCellsStatus: React.FC = () => {
  const columns: TColumn[] = useMemo(
    () => [
      {
        field: "id",
        title: "id"
      },
      {
        field: "fullName",
        title: "Full name",
        cellIcon: <IconSetting1 />
      },
      {
        field: "email",
        title: "email",
        cellTrailContent: <IconSetting1 />
      },
      {
        field: "statusText",
        title: "status",
        description: "P1-s",
        columnTypes: "status",
        cellParamsGetter: ({ data }): TCellParamsStatus => ({
          status: data.status,
          leadIcon: <IconSetting1 />
          isFilled: true // Can also be false; see the Status component
        }),
        iconDescription: <IconSetting1 />
      }
    ],
    []
  );

  const row = [
    {
      id: "1",
      fullName: "Palmerin de Lonfal",
      email: "palmslonfale@mail.com",
      statusText: "Error",
      status: "error"
    },
    {
      id: "2",
      fullName: "Damien de la Tour",
      email: "dmdtr@mail.com",
      statusText: "Warning",
      status: "warning"
    },
    {
      id: "3",
      fullName: "Auguste de Montferrand",
      email: "ogmonfer@fmail.com",
      statusText: "Success",
      status: "success"
    }
  ];

  return <Table title="Cells" columns={columns} rowData={row} />;
};
```

| Parameter | Type                    | Default      | Required | Description                               |
| --------- | ----------------------- | ------------ | -------- | ----------------------------------------- |

## `tags` cell type

#### `cellParamsGetter`

Configure `tags` cells with `cellParamsGetter`. It can receive `TCellTagsParamsGetterParams` and return per-cell settings, or you can pass one `TCellParamsTags` object for every cell.
Unlike `status`, `tags` does not require `cellParamsGetter`. Pass a string array in the row when no extra Tag properties are needed. For other shapes, such as an object array, use `valueGetter` to transform the value.

```tsx
export type TCellTagsParamsGetterParams<TData = any, TValue = string[]> = {
  value: TValue | null | undefined;
  node: IRowNode<TData> | null;
  data: TData | undefined;
  column: TColumn<TData, TValue>;
};

interface IColumnTags<TData = any, TValue = string>
  extends Omit<IColumnBasic<TData, TValue>, "columnTypes"> {
  columnTypes: "tags"; // The default column type is basic
  cellParamsGetter?: TCellParamsTagsGetter | TCellParamsTags;
}

export type TCellParamsTagsGetter<TData = any, TValue = string> = (
  params: TCellTagsParamsGetterParams<TData, TValue>
) => TCellParamsTags;

export type TCellParamsTags = {
  leadIcon?: JSX.Element;
};
```

```tsx
export const TableCellsTags: React.FC = () => {
  const columns: TColumn[] = useMemo(
    () => [
      {
        field: "id",
        title: "id"
      },
      {
        field: "fullName",
        title: "Full name",
        cellIcon: <IconSetting1 />
      },
      {
        field: "email",
        title: "email",
        cellTrailContent: <IconSetting1 />
      },
      {
        field: "tags",
        title: "Tags",
        columnTypes: "tags",
        cellParamsGetter: ({ data }): TCellParamsTags => ({
          leadIcon: data.leadIcon
        })
      },
      {
        field: "username",
        title: "valueGetter example",
        columnTypes: "tags",
        valueGetter: ({ data }) =>
          data.username.map(
            (username: { usernames: string[] }) => username.usernames
          )
      }
    ],
    []
  );

  const row = [
    {
      id: "1",
      username: [
        {
          usernames: "first"
        }
      ],
      tags: ["first", "second", "third"],
      leadIcon: <IconSetting1 />,
      fullName: "Palmerin de Lonfal",
      email: "palmslonfale@mail.com"
    },
    {
      id: "2",
      username: [
        {
          usernames: "first"
        }
      ],
      tags: ["first", "second", "third"],
      fullName: "Damien de la Tour",
      email: "damiendelatur@fmail.com"
    }
  ];

  return <Table title="Tag cells" columns={columns} rowData={row} />;
};
```

## `rating` cell type

Set `columnTypes: "rating"` to render a Rating component in a cell. Customize its appearance with `cellParamsGetter`. The default rating uses size `l` and the `blue` view.

#### `cellParamsGetter`

As with other cell types, the function receives `TCellRatingParamsGetterParams` and returns `TCellParamsRating`. To use the same properties for all cells, omit the function arguments and return `TCellParamsRating`.

```tsx
export type TCellRatingParamsGetterParams<TData = any, TValue = string> = {
  value: number | null | undefined;
  node: IRowNode<TData> | null;
  data: TData | undefined;
  column: TColumn<TData, TValue>;
};

export interface IColumnRating<TData = any>
  extends Omit<IColumnBasic<TData>, "columnTypes"> {
  columnTypes: "rating"; // The default column type is basic
  cellParamsGetter?: TCellParamsRatingGetter | TCellParamsRating; // Resolves Rating component parameters
}

type TCellParamsRatingGetter<TData = any> = (
  params: TCellRatingParamsGetterParams<TData>
) => TCellParamsRating;

export const ratingViewType = ["blue", "teal", "amber", "red"] as const;

export const buttonSize = ["l", "m"] as const;

export type TRatingViewType = (typeof ratingViewType)[number];

export type TRatingSize = (typeof buttonSize)[number];

export type TCellParamsRating = {
  viewType: TRatingViewType;
  size: TRatingSize;
};
```

## `Progress` cell

This cell contains a [`ProgressBar`](?path=/docs/components-neutrons-progressbar--documentation). Set `columnType: "progress"` in the column definition and provide the required `variant`. Configure `variant` and other options with `cellParamsGetter`.

#### `cellParamsGetter`

`cellParamsGetter` can receive `TCellProgressParamsGetterParams` and return per-cell settings. When all cells share the same properties, pass a `TCellParamsProgress` object instead.

```tsx
type TCellProgressParamsGetterParams<TData, TValue> = {
  value: TValue | null | undefined;
  node: IRowNode<TData> | null;
  data: TData | undefined;
  column: TColumn<TValue>;
};

type TCellParamsProgressGetter<TData = any, TValue = number> = (
  params: TCellProgressParamsGetterParams<TData, TValue>
) => TCellParamsProgress;

interface IColumnProgress<TData, TValue>
  extends Omit<IColumnBasic<TData, TValue>, "columnTypes"> {
  columnTypes: "progress"; // The default column type is basic
  cellParamsGetter?:
    | TCellParamsProgressGetter<TData, TValue>
    | TCellParamsProgress;
}

export type TProgressBarVariant = "info" | "success" | "warning" | "error";

export type TCellParamsProgress = {
  label?: string;
  variant: TProgressBarVariant;
  isText?: boolean;
};
```

## `number` cell type

## `Avatar` cell

Set `columnTypes: "avatar"` to render an [`Avatar`](?path=/docs/components-neutrons-avatar--documentation) in a cell.

#### cellParamsGetter

Configure avatar cells through `cellParamsGetter`, a function that receives `TCellAvatarParamsGetterParams` and can customize each cell independently.
Unlike other cell types, avatar `cellParamsGetter` must be a function; it cannot be a component-properties object.

```tsx
export interface IColumnAvatar<TData, TValue>
  extends Omit<IColumnBasic<TData, TValue>, "columnTypes"> {
  columnTypes: "avatar"; // The default column type is basic
  cellParamsGetter: TCellParamsAvatarGetter<TData, TValue>; // Resolves component parameters
}

export type TCellAvatarParamsGetterParams<TData, TValue> = {
  value: TValue | null | undefined;
  node: IRowNode<TData> | null;
  data: TData | undefined;
  column: TColumn<TData, TValue>;
};

type TCellParamsAvatarGetter<TData, TValue> = (
  params: TCellAvatarParamsGetterParams<TData, TValue>
) => TCellParamsTableAvatar;

const AVATAR_STATUS = ["online", "offline", "busy", "disabled"] as const;
type TAvatarStatus = (typeof AVATAR_STATUS)[number];

export type TCellParamsTableAvatar = {
  isDisabled?: boolean;
  image?: string;
  text?: string;
  icon?: JSX.Element;
  status?: TAvatarStatus;
  title?: string;
  description?: string;
};
```

```tsx
export const TableCellsAvatar: React.FC = () => {

        title: "id"
```
