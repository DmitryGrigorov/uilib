# Sorting

This page explains how to sort data and customize sorting behavior.

## Enable sorting

Set `isSortable` on a column and define its sorting options with `sortMenuItems`.

```tsx
const columns = useMemo(
  () => [
    {
      field: "user",
      isSortable: true,
      sortMenuItems: [
        {
          options: [
            { title: "Asc", nameSort: "asc" },
            { title: "Desc", nameSort: "desc" }
          ]
        }
      ]
    },
    {
      field: "username",
      isSortable: true,
      sortMenuItems: [
        {
          options: [
            { title: "Asc", nameSort: "asc" },
            { title: "Desc", nameSort: "desc" }
          ]
        }
      ]
    },
    { field: "age" }
  ],
  []
);

<Table columns={columns} />;
```

You can enable sorting for every column through [`defaultColumn`](?path=/docs/tables-default-column-settings--documentation).

```tsx
const defaultColumn = {
  isSortable: true,
  sortMenuItems: [
    {
      options: [
        { title: "Asc", nameSort: "asc" },
        { title: "Desc", nameSort: "desc" }
      ]
    }
  ]
};

const columns = useMemo(
  () => [
    {
      field: "user"
    },
    {
      field: "username"
    },
    { field: "age" }
  ],
  []
);

<Table columns={columns} defaultColumn={defaultColumn} />;
```

## Configure sorting

Configure sorting at the column level by providing a `comporator` function in `columns` or `defaultColumn`.

```ts
comporator = (
  valueA: TValue,
  valueB: TValue,
  nodeA: IRowNode<TData>,
  nodeB: IRowNode<TData>,
  nameSort: string
) => number;
```

- `valueA`, `valueB`: values to compare.
- `nodeA`, `nodeB`: the corresponding [row objects](?path=/docs/tables-row-object--documentation), useful when sorting needs additional row information.
- `nameSort`: the sorting option name defined in `sortMenuItems`.

Function result:

- `0`: `valueA` equals `valueB`.
- `>0`: sort `valueA` after `valueB`.
- `<0`: sort `valueA` before `valueB`.

## Example

```tsx
import { Table, TColumn } from "@dmitrygrigorov/table";

export const TableSorting: React.FC = () => {
  const columns: TColumn[] = useMemo(
    () => [
      {
        field: "id"
      },
      {
        field: "name",
        isSortable: true,
        sortMenuItems: [
          {
            options: [
              {
                title: "A to Z",
                nameSort: "asc"
              },
              {
                title: "Z to A",
                nameSort: "desc"
              }
            ]
          }
        ],
        comparator: (valueA, valueB, nameSort) => {
          if (nameSort === "asc") {
            return valueA > valueB ? 1 : -1;
          }
          if (nameSort === "desc") {
            return valueA < valueB ? -1 : 1;
          }
          return 0;
        }
      },
      {
        field: "username"
      },
      {
        field: "email"
      },
      {
        field: "address.city",
        title: "city"
      }
    ],
    []
  );

  const [comments, setComments] = useState<any[]>([]);

  const loadData = async (): Promise<void> => {
    const response = await fetch("https://jsonplaceholder.typicode.com/users");
    const data = (await response.json()) as any[];
    setComments(data);
  };

  useEffect(() => {
    loadData();
  }, []);

  return (
    <Table
      columns={columns}
      rowData={comments}
      size="s"
      defaultColumn={{ isSortable: true }}
    />
  );
};
```

### sortMenuItems properties

| Parameter | Type                   | Default | Required | Description         |
| --------- | ---------------------- | ------- | -------- | ------------------- |
| title     | `string`               | -       | -        | Sorting group title |
| options   | `ITableSortMenuOption` | -       | ✓        | Sorting options     |

### ITableSortMenuOption properties

| Parameter | Type     | Default | Required | Description                  |
| --------- | -------- | ------- | -------- | ---------------------------- |
| title     | `string` | -       | ✓        | Sorting label or description |
| nameSort  | `string` | -       | ✓        | Sorting option name          |
