# Pagination

## Pagination divides rows into pages so large datasets remain easy to browse.

Set `isPagination` to enable pagination.

```tsx
export const TablePagination: React.FC = () => {
  const columns = useMemo(
    () => [
      {
        field: "postId"
      },
      {
        field: "id"
      },
      {
        field: "name"
      },
      {
        field: "email"
      }
    ],
    []
  );

  const [comments, setComments] = useState<any[]>([]);

  const loadData = async (): Promise<void> => {
    const response = await fetch(
      "https://jsonplaceholder.typicode.com/comments"
    );
    const data = (await response.json()) as any[];
    setComments(data);
  };

  useEffect(() => {
    loadData();
  }, []);

  return (
    <Table
      columns={columns}
      rowData={comments}
      paginationPageSize={10}
      isPagination
      size="s"
    />
  );
};
```

## Pagination behavior

Tables support client-side pagination with `rowModel="client"` and server-side or manual pagination with `rowModel="server"`. See [Server-side and custom pagination](#server-side-and-custom-pagination). The default is `rowModel="client"`.

## Items per page

Set `isPaginationRowPerPage` to let users change how many items appear per page. The default choices are 10, 20, and 30; provide a new array through `paginationCounts` to customize them.

```tsx
export const TablePagination: React.FC = () => {
  const columns = useMemo(
    () => [
      {
        field: "postId"
      },
      {
        field: "id"
      },
      {
        field: "name"
      },
      {
        field: "email"
      }
    ],
    []
  );

  const [comments, setComments] = useState<any[]>([]);

  const loadData = async (): Promise<void> => {
    const response = await fetch(
      "https://jsonplaceholder.typicode.com/comments"
    );
    const data = (await response.json()) as any[];
    setComments(data);
  };

  useEffect(() => {
    loadData();
  }, []);

  return (
    <Table
      columns={columns}
      rowData={comments}
      paginationPageSize={5}
      isPagination
      size="s"
      isPaginationRowPerPage
      paginationCounts={[5, 15, 20]}
    />
  );
};
```

## Server-side and custom pagination

You can control the entire pagination flow by setting `rowModel="server"`.
The example below listens to `onPaginationChange` to update the current page information.
Pagination API methods work with both `rowModel="server"` and `rowModel="client"`; their reference appears below.

```tsx
export const TablePaginationServer: React.FC = (props) => {
  const columns = useMemo(
    () => [
      {
        field: "postId"
      },
      {
        field: "id"
      },
      {
        field: "name"
      },
      {
        field: "email"
      }
    ],
    []
  );
  const [page, setPage] = useState(1);
  const [rowPerPage, setRowPerPage] = useState(5);
  const [comments, setComments] = useState<any[]>([]);

  const handlePaginationChange = (params: IPaginationChangeParams): void => {
    setPage(params.newPage);
    setRowPerPage(params.rowPerPage);
  };

  const loadData = async (): Promise<void> => {
    const response = await fetch(
      `https://jsonplaceholder.typicode.com/comments?_start=${
        (page - 1) * rowPerPage
      }&_limit=${rowPerPage}`
    );
    const data = (await response.json()) as any[];
    setComments(data);
  };

  useEffect(() => {
    loadData();
  }, [page, rowPerPage]);
  return (
    <Table
      rowModel="server"
      columns={columns}
      rowData={comments}
      onPaginationChange={handlePaginationChange}
      paginationPageSize={rowPerPage}
      paginationAllRowCount={500}
      isPagination
    />
  );
};
```

## Available properties

| Property               | Type                                        | Description                                                                                      |
| ---------------------- | ------------------------------------------- | ------------------------------------------------------------------------------------------------ |
| isPagination           | `boolean`                                   | Whether pagination is displayed                                                                  |
| isPaginationRowPerPage | `boolean`                                   | Whether the rows-per-page menu is displayed                                                      |
| paginationPageSize     | `number`                                    | Number of displayed rows                                                                         |
| paginationCounts       | `Array<number>`                             | Values available in the rows-per-page selector                                                   |
| onPaginationChange     | `(params: IPaginationChangeParams) => void` | Callback receiving `newPage`, the new page number, and `rowPerPage`, the number of rows per page |
| paginationAllRowCount  | `number`                                    | Total row count. **Used when `rowModel="server"`**                                               |
| paginationCurrentPage  | `number`                                    | Initial pagination page                                                                          |

## API methods

| Property                   | Type                     | Description                                                             |
| -------------------------- | ------------------------ | ----------------------------------------------------------------------- |
| paginationGetPageSize      | `() => number`           | Returns the number of rows displayed per page                           |
| paginationGetCurrentPage   | `() => number`           | Returns the current page number                                         |
| paginationGetTotalPages    | `() => number`           | Returns the total number of pages                                       |
| paginationGetRowCount      | `() => number`           | Returns the total number of rows                                        |
| paginationGoToPage         | `(page: number) => void` | Goes to a page, or the last page when the requested page does not exist |
| paginationGoToNextPage     | `() => void`             | Goes to the next page                                                   |
| paginationGoToPreviousPage | `() => void`             | Goes to the previous page                                               |
| paginationGoToFirstPage    | `() => void`             | Goes to the first page                                                  |
| paginationGoToLastPage     | `() => void`             | Goes to the last page                                                   |

## Additional types

```ts
export interface IPaginationChangeParams {
  newPage: number;
  rowPerPage: number;
}
```
